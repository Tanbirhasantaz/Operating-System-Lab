#!/bin/bash

#Max Number Check
text="Hello World!"
length=${#text}
echo "length: $length"
read -p "Number1: " n1
read -p "Number2: " n2
read -p "number3: " n3
if [[ n1 -ge n2 ]]
then
	if [[ n1 -ge n3 ]]
	then 
		echo "n1 is max"
	else 
		echo "n3 is max"
	fi
else 
	if [[ n2 -ge n3 ]]
	then
		echo "n2 is max"
	else 
		echo "n3 is max"
	fi

fi


#Even/Odd Number Check
read -p "Enter a Number: " n
if [[ $(( n%2 )) -eq 1 ]]
then 
	echo "odd"
else 
	echo "even"
fi

#For Loop Implementation
sum=0
for (( i=1; i<=10; i++ ));
do
	#echo "i: $i"
	(( sum+=$i ))
done
echo "sum: $sum"

#While Loop Implementation
j=1
while [ $j -le 10 ];
do 
	echo "j: $j"
	(( j++ ))

done
